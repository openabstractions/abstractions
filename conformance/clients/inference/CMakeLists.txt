cmake_minimum_required(VERSION 3.16)
project(inference_consumer LANGUAGES CXX)
find_package(abstraction_facade CONFIG REQUIRED)
find_package(abstraction_inference_api CONFIG REQUIRED)
add_executable(inference_consumer main.cpp)
target_compile_features(inference_consumer PRIVATE cxx_std_17)
target_link_libraries(inference_consumer PRIVATE abstraction::facade_client abstraction::inference_api)
add_executable(inference_waiting_test waiting.cpp)
target_compile_features(inference_waiting_test PRIVATE cxx_std_17)
target_compile_definitions(inference_waiting_test PRIVATE OA_INFERENCE_WAITING_REPAIR)
target_link_libraries(inference_waiting_test PRIVATE abstraction::facade_client abstraction::inference_api)
enable_testing()
add_test(NAME inference_waiting COMMAND inference_waiting_test)
