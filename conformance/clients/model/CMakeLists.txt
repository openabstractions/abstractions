cmake_minimum_required(VERSION 3.16)
project(model_consumer LANGUAGES CXX)
find_package(abstraction_model CONFIG REQUIRED)
option(MODEL_DIRECT_ONLY "Build independently installed model client only" OFF)
if(NOT MODEL_DIRECT_ONLY)
    find_package(abstraction_facade CONFIG REQUIRED)
    add_executable(model_consumer client.cpp)
    target_link_libraries(model_consumer PRIVATE abstraction::facade_client)
endif()
add_executable(model_direct client.cpp)
target_compile_definitions(model_direct PRIVATE MODEL_DIRECT=1)
target_link_libraries(model_direct PRIVATE abstraction::model_client)

option(MODEL_GENERIC "Build model plus independent generic resolver" OFF)
if(MODEL_GENERIC)
    find_package(abstraction_facade_resolution CONFIG REQUIRED)
    add_executable(model_generic generic.cpp)
    target_link_libraries(model_generic PRIVATE abstraction::model_client abstraction::facade_resolution)
endif()
